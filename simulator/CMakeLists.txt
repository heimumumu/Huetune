cmake_minimum_required(VERSION 3.20)
project(controller_ui_sim C CXX)
set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> ar qc <TARGET> <LINK_FLAGS> <OBJECTS>")
set(CMAKE_C_ARCHIVE_FINISH "")
set(CMAKE_CXX_STANDARD 17)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory(../firmware/managed_components/lvgl__lvgl lvgl)
get_target_property(LV_SOURCES lvgl SOURCES)
list(FILTER LV_SOURCES EXCLUDE REGEX "\\.S$")
set_property(TARGET lvgl PROPERTY SOURCES ${LV_SOURCES})
file(GLOB FONTS ../firmware/generated/mc_font_*.c)
add_executable(controller_ui_sim main.cpp ../firmware/main/ui.cpp ${FONTS})
target_include_directories(controller_ui_sim PRIVATE ../firmware/main)
target_compile_definitions(controller_ui_sim PRIVATE LV_LVGL_H_INCLUDE_SIMPLE)
target_compile_options(controller_ui_sim PRIVATE -UNDEBUG)
target_link_libraries(controller_ui_sim PRIVATE lvgl)
